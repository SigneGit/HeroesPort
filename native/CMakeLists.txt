cmake_minimum_required(VERSION 3.25)
project(DDHeroes LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT WIN32 OR NOT CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native host needs Windows x64 and Clang targeting the MSVC ABI. Portable MinGW builds only the analysis tools.")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Use clang/clang++ from a Visual Studio x64 developer environment.")
endif()
option(HEROES_LAUNCHER_ONLY "Build the ISO import launcher without game files" OFF)
if(HEROES_LAUNCHER_ONLY)
    include(FetchContent)
    set(FETCHCONTENT_SOURCE_DIR_TOMLPLUSPLUS "${CMAKE_CURRENT_SOURCE_DIR}/../local/native-deps/tomlplusplus" CACHE PATH "Pinned dependency")
    FetchContent_Declare(tomlplusplus SOURCE_DIR "${FETCHCONTENT_SOURCE_DIR_TOMLPLUSPLUS}")
    FetchContent_MakeAvailable(tomlplusplus)
    add_executable(HeroesLauncher WIN32 launcher/main.cpp)
    target_compile_definitions(HeroesLauncher PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(HeroesLauncher PRIVATE tomlplusplus::tomlplusplus comdlg32 shell32)
    return()
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/default.xbe" OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config/heroes.toml")
    message(FATAL_ERROR "Import the user image, then run node tools/prepare-native.mjs from the repository root.")
endif()
set(HEROES_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../local/native-deps")
set(ZYAN_ZYCORE_PATH "${HEROES_DEPS}/zycore" CACHE PATH "Pinned Zycore source")
foreach(dep zydis tomlplusplus imgui sdl3)
    if(IS_DIRECTORY "${HEROES_DEPS}/${dep}")
        string(TOUPPER "${dep}" dep_upper)
        set(FETCHCONTENT_SOURCE_DIR_${dep_upper} "${HEROES_DEPS}/${dep}" CACHE PATH "Pinned dependency")
    endif()
endforeach()
include("${CMAKE_CURRENT_SOURCE_DIR}/../third_party/xboxdecomp/cmake/XbeRecomp.cmake")
file(GLOB XBE_HOST_SOURCES CONFIGURE_DEPENDS src/common/*.cpp)
# Route only the SDK entry point through our portable preparation dispatcher.
set_property(SOURCE "${XBE_SDK_DIR}/runtime/app/main.cpp" APPEND PROPERTY
    COMPILE_DEFINITIONS wWinMain=HeroesGameMain)
add_xbe_image(Heroes default.xbe heroes.toml 64 heroes)
target_link_libraries(Heroes PRIVATE shell32)
# The SDK accepts a development DLL only when newer than its host executable.
# An unchanged import library can otherwise let Ninja keep an older DLL after
# a host-only change, triggering an unnecessary player-side compilation.
set_property(TARGET Heroes_code APPEND PROPERTY LINK_DEPENDS "$<TARGET_FILE:Heroes>")
add_executable(HeroesLauncher WIN32 launcher/main.cpp)
target_compile_definitions(HeroesLauncher PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(HeroesLauncher PRIVATE tomlplusplus::tomlplusplus comdlg32 shell32)
